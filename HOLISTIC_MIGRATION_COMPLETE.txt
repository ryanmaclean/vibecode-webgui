USAGE EXAMPLES

4. New Script Template:
   #!/usr/bin/env python3
   from lib.vibecode_common import init_vibecode_script, tracer
   
   logger, config, metrics, shutdown = init_vibecode_script('my_script')
   
   @tracer.wrap(service='my-service', resource='work')
   def do_work():
       logger.info("Working...")
       if metrics:
           metrics.increment('work.done')
   
   if __name__ == '__main__':
       with tracer.trace('session', service='my-service'):
           do_work()
